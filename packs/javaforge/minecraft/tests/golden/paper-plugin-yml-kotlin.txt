# BetterChat

A Paper plugin for Minecraft 1.21.8, written in Kotlin (Java 21).

## Build

The project has no Gradle wrapper yet. Create it once with an installed Gradle:

```sh
gradle wrapper
```

Then build the plugin:

```sh
./gradlew build
```

The plugin jar is `build/libs/better-chat-1.0.0-SNAPSHOT.jar`. Copy it into the `plugins/` folder of your server.

## Test server

`./gradlew runServer` (or the **Run Paper Server** run configuration in Forge) downloads Paper 1.21.8 with [run-paper](https://github.com/jpenilla/run-gradle), builds the plugin and starts a server in `run/` with the plugin installed. Accept the Minecraft EULA in `run/eula.txt` on the first start.

## Structure

- `src/main/kotlin/dev/alex/betterchat/BetterChat.kt`: the main class
- `src/main/kotlin/dev/alex/betterchat/command/HelloCommand.kt`: the `/hello` command
- `src/main/kotlin/dev/alex/betterchat/listener/JoinListener.kt`: greets players when they join
- `src/main/resources/paper-plugin.yml`: the plugin descriptor

The Kotlin standard library is bundled into the plugin jar with Shadow.
==== build.gradle.kts ====
plugins {
    kotlin("jvm") version "2.2.20"
    id("com.gradleup.shadow") version "8.3.6"
    id("xyz.jpenilla.run-paper") version "2.3.1"
}

group = "dev.alex"
version = "1.0.0-SNAPSHOT"

repositories {
    mavenCentral()
    maven("https://repo.papermc.io/repository/maven-public/")
}

dependencies {
    compileOnly("io.papermc.paper:paper-api:1.21.8-R0.1-SNAPSHOT")
}

kotlin {
    jvmToolchain(21)
}

tasks {
    processResources {
        val props = mapOf("version" to version)
        inputs.properties(props)
        filteringCharset = "UTF-8"
        filesMatching("paper-plugin.yml") {
            expand(props)
        }
    }
    shadowJar {
        archiveClassifier.set("")
    }
    build {
        dependsOn(shadowJar)
    }
    runServer {
        // The Paper version of the test server (./run).
        minecraftVersion("1.21.8")
    }
}

==== src/main/kotlin/dev/alex/betterchat/BetterChat.kt ====
package dev.alex.betterchat

import dev.alex.betterchat.command.HelloCommand
import dev.alex.betterchat.listener.JoinListener
import io.papermc.paper.plugin.lifecycle.event.types.LifecycleEvents
import org.bukkit.plugin.java.JavaPlugin

class BetterChat : JavaPlugin() {

    override fun onEnable() {
        server.pluginManager.registerEvents(JoinListener(), this)
        // paper-plugin.yml has no commands section: commands are registered with Brigadier.
        lifecycleManager.registerEventHandler(LifecycleEvents.COMMANDS) { event ->
            event.registrar().register("hello", "Says hello", HelloCommand())
        }
        logger.info("BetterChat enabled")
    }

    override fun onDisable() {
        logger.info("BetterChat disabled")
    }
}
==== src/main/kotlin/dev/alex/betterchat/command/HelloCommand.kt ====
package dev.alex.betterchat.command

import io.papermc.paper.command.brigadier.BasicCommand
import io.papermc.paper.command.brigadier.CommandSourceStack
import net.kyori.adventure.text.Component
import net.kyori.adventure.text.format.NamedTextColor

/** `/hello`: greets whoever runs it. */
class HelloCommand : BasicCommand {

    override fun execute(source: CommandSourceStack, args: Array<out String>) {
        source.sender.sendMessage(Component.text("Hello, ${source.sender.name}!", NamedTextColor.GREEN))
    }
}
==== src/main/kotlin/dev/alex/betterchat/listener/JoinListener.kt ====
package dev.alex.betterchat.listener

import net.kyori.adventure.text.Component
import net.kyori.adventure.text.format.NamedTextColor
import org.bukkit.event.EventHandler
import org.bukkit.event.Listener
import org.bukkit.event.player.PlayerJoinEvent

class JoinListener : Listener {

    @EventHandler
    fun onPlayerJoin(event: PlayerJoinEvent) {
        event.player.sendMessage(Component.text("Welcome, ${event.player.name}!", NamedTextColor.YELLOW))
    }
}
