# BetterChat

Nicer chat messages

A Paper plugin for Minecraft 1.21.11, written in Java (Java 21).

## Build

The project has no Gradle wrapper yet. Create it once with an installed Gradle (9 or newer is recommended for Shadow 9 and run-paper 3):

```sh
gradle wrapper
```

Then build the plugin:

```sh
./gradlew build
```

The plugin jar is `build/libs/better-chat-1.0.0-SNAPSHOT.jar`. Copy it into the `plugins/` folder of your server.

## Test server

`./gradlew runServer` (or the **Run Paper Server** run configuration in Forge) downloads Paper 1.21.11 with [run-paper](https://github.com/jpenilla/run-gradle), builds the plugin and starts a server in `run/` with the plugin installed. Accept the Minecraft EULA in `run/eula.txt` on the first start.

## Structure

- `src/main/java/com/example/betterchat/BetterChat.java`: the main class
- `src/main/java/com/example/betterchat/command/HelloCommand.java`: the `/hello` command
- `src/main/java/com/example/betterchat/listener/JoinListener.java`: greets players when they join
- `src/main/resources/plugin.yml`: the plugin descriptor
==== build.gradle.kts ====
plugins {
    java
    id("xyz.jpenilla.run-paper") version "3.1.0"
}

group = "com.example"
version = "1.0.0-SNAPSHOT"
description = "Nicer chat messages"

repositories {
    mavenCentral()
    maven("https://repo.papermc.io/repository/maven-public/")
}

dependencies {
    compileOnly("io.papermc.paper:paper-api:1.21.11-R0.1-SNAPSHOT")
}

java {
    toolchain.languageVersion.set(JavaLanguageVersion.of(21))
}

tasks {
    compileJava {
        options.encoding = "UTF-8"
        options.release.set(21)
    }
    processResources {
        val props = mapOf("version" to version)
        inputs.properties(props)
        filteringCharset = "UTF-8"
        filesMatching("plugin.yml") {
            expand(props)
        }
    }
    runServer {
        // The Paper version of the test server (./run).
        minecraftVersion("1.21.11")
    }
}

==== src/main/java/com/example/betterchat/BetterChat.java ====
package com.example.betterchat;

import com.example.betterchat.command.HelloCommand;
import com.example.betterchat.listener.JoinListener;
import org.bukkit.command.PluginCommand;
import org.bukkit.plugin.java.JavaPlugin;

public final class BetterChat extends JavaPlugin {

    @Override
    public void onEnable() {
        getServer().getPluginManager().registerEvents(new JoinListener(), this);
        PluginCommand hello = getCommand("hello");
        if (hello != null) {
            hello.setExecutor(new HelloCommand());
        }
        getLogger().info("BetterChat enabled");
    }

    @Override
    public void onDisable() {
        getLogger().info("BetterChat disabled");
    }
}
==== src/main/java/com/example/betterchat/command/HelloCommand.java ====
package com.example.betterchat.command;

import net.kyori.adventure.text.Component;
import net.kyori.adventure.text.format.NamedTextColor;
import org.bukkit.command.Command;
import org.bukkit.command.CommandExecutor;
import org.bukkit.command.CommandSender;

/** {@code /hello}: greets whoever runs it. */
public final class HelloCommand implements CommandExecutor {

    @Override
    public boolean onCommand(CommandSender sender, Command command, String label, String[] args) {
        sender.sendMessage(Component.text("Hello, " + sender.getName() + "!", NamedTextColor.GREEN));
        return true;
    }
}
==== src/main/java/com/example/betterchat/listener/JoinListener.java ====
package com.example.betterchat.listener;

import net.kyori.adventure.text.Component;
import net.kyori.adventure.text.format.NamedTextColor;
import org.bukkit.event.EventHandler;
import org.bukkit.event.Listener;
import org.bukkit.event.player.PlayerJoinEvent;

public final class JoinListener implements Listener {

    @EventHandler
    public void onPlayerJoin(PlayerJoinEvent event) {
        String name = event.getPlayer().getName();
        event.getPlayer().sendMessage(Component.text("Welcome, " + name + "!", NamedTextColor.YELLOW));
    }
}
