# Better Chests

Bigger chests

A Fabric mod for Minecraft 26.3 (Fabric Loader 0.20.1), written in Java (Java 25).

Minecraft 26.1 and newer ship without obfuscation, so the code uses Mojang's names directly.

## Build

The project has no Gradle wrapper yet. Create it once with an installed Gradle 9 or newer:

```sh
gradle wrapper
```

Then build the mod (Gradle downloads a JDK 25 with the foojay toolchain resolver when none is installed):

```sh
./gradlew build
```

The mod jar is `build/libs/better_chests-1.0.0.jar`. Copy it into the `mods/` folder of a Minecraft 26.3 installation with Fabric and Fabric API.

## Run

- `./gradlew runClient` (the **Minecraft Client** run configuration in Forge) starts the game with the mod.
- `./gradlew runServer` (**Minecraft Server**) starts a dedicated server in `run/`. Accept the Minecraft EULA in `run/eula.txt` on the first start.

## Structure

- `src/main/java/com/example/betterchests/BetterChests.java`: the main class
- `src/client/java/com/example/betterchests/client/BetterChestsClient.java`: the client initializer
- `src/main/java/com/example/betterchests/registry/`: the example item and block
- `src/main/resources/assets/better_chests/`: translations, models and textures
- `src/main/resources/fabric.mod.json`: the mod's metadata
- `src/main/resources/better_chests.mixins.json`: the mixin configuration, with the example mixins in the `mixin` package
==== build.gradle ====
plugins {
    id 'net.fabricmc.fabric-loom' version "${loom_version}"
}

version = project.mod_version
group = project.maven_group

base {
    archivesName = project.archives_base_name
}

repositories {
    // Loom adds the repositories of Minecraft, Fabric and their libraries. Add the
    // repositories of other mods you depend on here.
}

loom {
    // Client code lives in src/client and cannot be loaded on a dedicated server.
    splitEnvironmentSourceSets()

    mods {
        'better_chests' {
            sourceSet sourceSets.main
            sourceSet sourceSets.client
        }
    }
}

dependencies {
    minecraft "com.mojang:minecraft:${project.minecraft_version}"
    implementation "net.fabricmc:fabric-loader:${project.loader_version}"
    implementation "net.fabricmc.fabric-api:fabric-api:${project.fabric_api_version}"
}

processResources {
    def version = project.version
    inputs.property 'version', version

    filesMatching('fabric.mod.json') {
        expand 'version': version
    }
}

java {
    toolchain.languageVersion = JavaLanguageVersion.of(25)
    withSourcesJar()
}

tasks.withType(JavaCompile).configureEach {
    options.encoding = 'UTF-8'
    options.release = 25
}

==== src/main/java/com/example/betterchests/BetterChests.java ====
package com.example.betterchests;

import net.fabricmc.api.ModInitializer;
import net.minecraft.resources.Identifier;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import com.example.betterchests.registry.ModBlocks;
import com.example.betterchests.registry.ModItems;

public class BetterChests implements ModInitializer {
    public static final String MOD_ID = "better_chests";

    // Named after the mod id, so the log shows which mod wrote a line.
    public static final Logger LOGGER = LoggerFactory.getLogger(MOD_ID);

    @Override
    public void onInitialize() {
        // Runs as soon as Minecraft is ready to load mods: register content here.
        ModItems.initialize();
        ModBlocks.initialize();
        LOGGER.info("Hello from Better Chests!");
    }

    public static Identifier id(String path) {
        return Identifier.fromNamespaceAndPath(MOD_ID, path);
    }
}

==== src/main/java/com/example/betterchests/registry/ModBlocks.java ====
package com.example.betterchests.registry;

import java.util.function.Function;
import net.minecraft.core.Registry;
import net.minecraft.core.registries.BuiltInRegistries;
import net.minecraft.core.registries.Registries;
import net.minecraft.resources.ResourceKey;
import net.minecraft.world.item.Item;
import net.minecraft.world.item.BlockItem;
import net.minecraft.world.level.block.Block;
import net.minecraft.world.level.block.state.BlockBehaviour;
import net.minecraft.world.level.block.SoundType;
import com.example.betterchests.BetterChests;

public final class ModBlocks {
    public static final Block EXAMPLE_BLOCK = register("example_block", Block::new,
            BlockBehaviour.Properties.of().strength(1.5f).sound(SoundType.STONE));

    private ModBlocks() {
    }

    /** Registers the block and its item. */
    public static Block register(String name, Function<BlockBehaviour.Properties, Block> factory, BlockBehaviour.Properties properties) {
        ResourceKey<Block> key = ResourceKey.create(Registries.BLOCK, BetterChests.id(name));
        Block block = Registry.register(BuiltInRegistries.BLOCK, key, factory.apply(properties.setId(key)));
        ModItems.register(name, p -> new BlockItem(block, p), new Item.Properties().useBlockDescriptionPrefix());
        return block;
    }

    /** Loads the class, which registers its blocks. */
    public static void initialize() {
    }
}
==== src/main/java/com/example/betterchests/registry/ModItems.java ====
package com.example.betterchests.registry;

import java.util.function.Function;
import net.minecraft.core.Registry;
import net.minecraft.core.registries.BuiltInRegistries;
import net.minecraft.core.registries.Registries;
import net.minecraft.resources.ResourceKey;
import net.minecraft.world.item.Item;
import com.example.betterchests.BetterChests;

public final class ModItems {
    public static final Item EXAMPLE_ITEM = register("example_item", Item::new, new Item.Properties());

    private ModItems() {
    }

    public static <T extends Item> T register(String name, Function<Item.Properties, T> factory, Item.Properties properties) {
        ResourceKey<Item> key = ResourceKey.create(Registries.ITEM, BetterChests.id(name));
        return Registry.register(BuiltInRegistries.ITEM, key, factory.apply(properties.setId(key)));
    }

    /** Loads the class, which registers its items. */
    public static void initialize() {
    }
}
