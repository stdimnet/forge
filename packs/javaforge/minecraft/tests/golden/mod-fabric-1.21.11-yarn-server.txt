# Better Chests

A Fabric mod for Minecraft 1.21.11 (Fabric Loader 0.20.1), written in Java (Java 21).

The code uses Fabric's Yarn names (1.21.11+build.6).

## Build

The project has no Gradle wrapper yet. Create it once with an installed Gradle 9 or newer:

```sh
gradle wrapper
```

Then build the mod (Gradle downloads a JDK 21 with the foojay toolchain resolver when none is installed):

```sh
./gradlew build
```

The mod jar is `build/libs/better_chests-1.0.0.jar`. Copy it into the `mods/` folder of a Minecraft 1.21.11 installation with Fabric and Fabric API.

## Run

- `./gradlew runServer` (**Minecraft Server**) starts a dedicated server in `run/`. Accept the Minecraft EULA in `run/eula.txt` on the first start.

## Structure

- `src/main/java/com/example/betterchests/BetterChests.java`: the main class
- `src/main/java/com/example/betterchests/BetterChestsServer.java`: the dedicated server initializer
- `src/main/java/com/example/betterchests/registry/`: the example item and block
- `src/main/resources/assets/better_chests/`: translations, models and textures
- `src/main/resources/fabric.mod.json`: the mod's metadata

plugins {

}

version = project.mod_version

base {
    archivesName = project.archives_base_name
}

repositories {
    // Loom adds the repositories of Minecraft, Fabric and their libraries. Add the
    // repositories of other mods you depend on here.
}

loom {
    mods {
        'better_chests' {
            sourceSet sourceSets.main
        }
    }
}

dependencies {
    minecraft "com.mojang:minecraft:${project.minecraft_version}"
    mappings "net.fabricmc:yarn:${project.yarn_mappings}:v2"
    modImplementation "net.fabricmc:fabric-loader:${project.loader_version}"
    modImplementation "net.fabricmc.fabric-api:fabric-api:${project.fabric_api_version}"
}

processResources {
    def version = project.version
    inputs.property 'version', version

    filesMatching('fabric.mod.json') {
        expand 'version': version
    }
}

java {
    toolchain.languageVersion = JavaLanguageVersion.of(21)
    withSourcesJar()
}

tasks.withType(JavaCompile).configureEach {
    options.encoding = 'UTF-8'
    options.release = 21
}

==== src/main/java/com/example/betterchests/BetterChests.java ====
package com.example.betterchests;

import net.fabricmc.api.ModInitializer;
import net.minecraft.util.Identifier;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import com.example.betterchests.registry.ModBlocks;
import com.example.betterchests.registry.ModItems;

public class BetterChests implements ModInitializer {
    public static final String MOD_ID = "better_chests";

    // Named after the mod id, so the log shows which mod wrote a line.
    public static final Logger LOGGER = LoggerFactory.getLogger(MOD_ID);

    @Override
    public void onInitialize() {
        // Runs as soon as Minecraft is ready to load mods: register content here.
        ModItems.initialize();
        ModBlocks.initialize();
        LOGGER.info("Hello from Better Chests!");
    }

    public static Identifier id(String path) {
        return Identifier.of(MOD_ID, path);
    }
}

==== src/main/java/com/example/betterchests/registry/ModBlocks.java ====
package com.example.betterchests.registry;

import java.util.function.Function;
import net.minecraft.registry.Registry;
import net.minecraft.registry.Registries;
import net.minecraft.registry.RegistryKeys;
import net.minecraft.registry.RegistryKey;
import net.minecraft.item.Item;
import net.minecraft.item.BlockItem;
import net.minecraft.block.Block;
import net.minecraft.block.AbstractBlock;
import net.minecraft.sound.BlockSoundGroup;
import com.example.betterchests.BetterChests;

public final class ModBlocks {
    public static final Block EXAMPLE_BLOCK = register("example_block", Block::new,
            AbstractBlock.Settings.create().strength(1.5f).sounds(BlockSoundGroup.STONE));

    private ModBlocks() {
    }

    /** Registers the block and its item. */
    public static Block register(String name, Function<AbstractBlock.Settings, Block> factory, AbstractBlock.Settings properties) {
        RegistryKey<Block> key = RegistryKey.of(RegistryKeys.BLOCK, BetterChests.id(name));
        Block block = Registry.register(Registries.BLOCK, key, factory.apply(properties.registryKey(key)));
        ModItems.register(name, p -> new BlockItem(block, p), new Item.Settings().useBlockPrefixedTranslationKey());
        return block;
    }

    /** Loads the class, which registers its blocks. */
    public static void initialize() {
    }
}
==== src/main/java/com/example/betterchests/registry/ModItems.java ====
package com.example.betterchests.registry;

import java.util.function.Function;
import net.minecraft.registry.Registry;
import net.minecraft.registry.Registries;
import net.minecraft.registry.RegistryKeys;
import net.minecraft.registry.RegistryKey;
import net.minecraft.item.Item;
import com.example.betterchests.BetterChests;

public final class ModItems {
    public static final Item EXAMPLE_ITEM = register("example_item", Item::new, new Item.Settings());

    private ModItems() {
    }

    public static <T extends Item> T register(String name, Function<Item.Settings, T> factory, Item.Settings properties) {
        RegistryKey<Item> key = RegistryKey.of(RegistryKeys.ITEM, BetterChests.id(name));
        return Registry.register(Registries.ITEM, key, factory.apply(properties.registryKey(key)));
    }

    /** Loads the class, which registers its items. */
    public static void initialize() {
    }
}
